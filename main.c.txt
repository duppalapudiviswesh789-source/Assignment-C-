#include <stdio.h>
#define MAX 5 
int main() {
    // first declaration 
    // 
    int ids[MAX];
    char  names[MAX][20];
    float prices[MAX];
    int quantities[MAX];
   //second part 
    int count = 0;
    int choice;
    int i ;
    int id;
    int pos;
    int newqty;
     do{
         printf("\n========================\n");
         printf(" PRODUCT INVENTORY SYSTEM\n");
         printf("=========================\n");
         printf(" 1.Add Product\n");
         printf(" 2.Search Product\n");
         printf(" 3.Display All Products\n");
         printf(" 4.Update quantity\n");
         printf(" 5.Exit\n");
         printf("=======================\n");
         printf(" ENTER YOUR CHOICE(1-5): ");
         scanf("%d",&choice);
         switch(choice){
             case 1 :
                  if ( count == MAX){
                       printf(" \n INVENTORY IS FULL\n");
                  }
             else{
                  printf(" Enter Product ID: ");
                  scanf("%d",&ids[count]);
                  printf(" Enter product name : ");
                 scanf(" %19[^\n]",names[count]);
                  printf(" Enter Product Price : ");
                 scanf("%f",&prices[count]);
                  printf(" Enter product Quantity :");
                 scanf("%d",&quantities[count]);
                 count++;
                 printf("\n Product Added Succsfully !\n");
                 }
                 break;
                 case 2 :
                 if(count ==0)
                  printf("\n INVENTORY IS EMPTY!1\n");
                  else {
                    printf("Enter ID to Search : ");
                    scanf("%d",&id);
                    pos = -1;
                    for(i = 0;i<count;i++){
                        if (ids[i] == id) {
                            pos = i;
                            break;
                        }
                    }             
                    if (pos == -1){
                        printf("Product not found");
                    }
                         else {
                            printf(" \n Product Found ");
                            printf("--------------------\n ");
                            printf(" ID        : %d\n",ids[pos]);
                            printf(" Name      : %s\n",names[pos]);
                            printf(" Price     : %.2f\n",prices[pos]);
                            printf(" Quantity  : %d\n",quantities[pos]);
                            printf(" -------------------\n");
                                  } }
                        break;
            case 3 :
            if (count == 0){
                printf("\n INVENTORY IS EMPTY");
            } 
            else {
                printf("\n%-8s%-25s%-12s%s\n","ID","NAME","PRICE","Quantity");
                printf("------------------------------------------\n");
                for(i = 0;i<count;i++){
                    printf("%-8d%-25s%-12.2f%d\n", ids[i], names[i], prices[i], quantities[i]);
                }
                }
                break;
                case 4 :
                if (count == 0 ){
                    printf("\n INVENTORY IS EMPTY \n");
                } else {
                    printf("Enter Product to Update : ");
                    scanf("%d",&id);
                    pos = -1;
                    for (i = 0; i < count; i++) {
                        if (ids[i] == id) {
                            pos = i;
                            break;
                        }
                    }
                    if (pos == -1){
                        printf("\n Product is not found!!\n");

                    }else {
                        printf("Current Quantity : %d\n",quantities[pos]);
                        printf(" Enter New Quantity : ");
                        scanf("%d",&newqty);
                        quantities[pos] = newqty;
                        printf("\n Quantity Updated Successfully \n");
                    }
                } break;
                case 5 : 
                printf("\n Exiting the program , GOOD BYE!!\n");
                break;
                default:
                printf("\n Invalid choice !! Enter 1-5\n");
            }               
              

                  
         
     }while(choice!=5);
    
    return 0;
}
